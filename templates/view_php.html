<?php
// user_controller.php

include_once("config.php");

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $username = $_POST['username'] ?? '';
    $email = $_POST['email'] ?? '';

    if ($username && $email) {
        $stmt = $pdo->prepare("INSERT INTO users (username, email) VALUES (?, ?)");
        $stmt->execute([$username, $email]);
        echo "User created!";
    } else {
        echo "Missing fields.";
    }
}
?>
